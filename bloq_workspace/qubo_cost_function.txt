```python
def bloq_pqc_cost_function(bitstring: str) -> float:
    """
    Custom Cost Function for Bloq Experiments Studio.
    bitstring: 15-character binary string representing ECU assignments.
    """
    x = [int(b) for b in bitstring]
    cost = 0.0
    linear = [450.0747, -49.7818, -49.0525, 152.6249, 153.5725, 450.0473, -49.5438, -49.741, 152.2452, 153.0042, 450.2508, -49.7343, -49.6918, -49.5242, -50.256]
    for i, c in enumerate(linear):
        cost += c * x[i]
    # Penalty for violating 1-algorithm-per-ECU constraint (3 ECUs x 5 algos)
    for e_idx in range(3):
        ecu_sum = sum(x[e_idx*5 : (e_idx+1)*5])
        cost += 50.0 * ((1 - ecu_sum) ** 2)
    return cost
```
